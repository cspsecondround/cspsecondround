void bubbleSort()
{
    bool f = 0;

    for (int i = 0; i < n - 1; i++)
    {
        f = 0;
        for (int j = 0; j < n - 1 - i; j++)
        {
            if (a[j] > a[j + 1])
            {
                swap(a[j + 1], a[j]);
                f = 1;
            }
        }

        if (!f)
        {
            return;
        }
    }
}

void merge(int l, int m, int r)
{
    int i = l, j = m + 1, k = l;
    while (i <= m && j <= r)
    {
        if (a[i] <= a[j])
            tmp[k++] = a[i++];
        else
            tmp[k++] = a[j++];
    }
    while (i <= m)
        tmp[k++] = a[i++];
    while (j <= r)
        tmp[k++] = a[j++];
    for (i = l; i <= r; i++)
        a[i] = tmp[i];
}

void merge_sort(int l, int r)
{
    if (l == r)
        return;
    int m = (l + r) / 2;
    merge_sort(l, m);
    merge_sort(m + 1, r);
    merge(l, m, r);
}

快速

void quickSort(int a[], int l, int r)
{
    int i, j, t, p;
    if (l > r)
        return;
    p = a[l];
    i = l;
    j = r;
    while (i != j)
    {
        while (a[j] >= p && i < j)
            j--;
        while (a[i] <= p && i < j)
            i++;
        if (i < j)
        {
            t = a[i];
            a[i] = a[j];
            a[j] = t;
        }
    }
    a[l] = a[i];
    a[i] = p;
    quickSort(a, l, i - 1);
    quickSort(a, i + 1, r);
}

Dijkstra（堆优化）

#include <bits/stdc++.h>

using namespace std;

const int INF = 1e18;

int n, m, s;
int dis[100005];
bool vis[100005];
vector<pair<int, int>> g[100005];   // {to, w}

void dijkstra()
{
    for (int i = 1; i <= n; i++)
        dis[i] = INF;
    dis[s] = 0;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, s});
    while (!pq.empty())
    {
        int u = pq.top().second;
        pq.pop();
        if (vis[u])
            continue;
        vis[u] = 1;
        for (auto &e : g[u])
        {
            int v = e.first, w = e.second;
            if (dis[v] > dis[u] + w)
            {
                dis[v] = dis[u] + w;
                pq.push({dis[v], v});
            }
        }
    }
}

int main()
{
    cin >> n >> m >> s;
    for (int i = 1; i <= m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        g[u].push_back({v, w});
    }
    dijkstra();
    for (int i = 1; i <= n; i++)
    {
        cout << dis[i] << ' ';
    }
    return 0;
}

SPFA

#include <bits/stdc++.h>

using namespace std;

const int INF = 1e18;

int n, m, s;
int dis[100005], cnt[100005];
bool inq[100005];
vector<pair<int, int>> g[100005];
queue<int> q;

bool spfa()
{
    for (int i = 1; i <= n; i++)
        dis[i] = INF;
    dis[s] = 0;
    q.push(s);
    inq[s] = 1;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        inq[u] = 0;
        for (auto &e : g[u])
        {
            int v = e.first, w = e.second;
            if (dis[v] > dis[u] + w)
            {
                dis[v] = dis[u] + w;
                cnt[v] = cnt[u] + 1;
                if (cnt[v] >= n)
                    return 0;   // 负环
                if (!inq[v])
                {
                    q.push(v);
                    inq[v] = 1;
                }
            }
        }
    }
    return 1;
}

int main()
{
    cin >> n >> m >> s;
    for (int i = 1; i <= m; i++)
    {
        int u, v, w;
        cin >> u >> v >> w;
        g[u].push_back({v, w});
    }
    if (!spfa())
    {
        cout << "负环";
        return 0;
    }
    for (int i = 1; i <= n; i++)
    {
        cout << dis[i] << ' ';
    }
    return 0;
}

Bellman-Ford

#include <bits/stdc++.h>

using namespace std;

const int INF = 1e18;

int n, m, s;
int dis[100005];

struct Edge
{
    int u, v, w;
} e[100005];

bool bellman()
{
    for (int i = 1; i <= n; i++)
        dis[i] = INF;
    dis[s] = 0;
    for (int i = 1; i <= n - 1; i++)
    {
        for (int j = 1; j <= m; j++)
        {
            int u = e[j].u, v = e[j].v, w = e[j].w;
            if (dis[u] != INF && dis[v] > dis[u] + w)
                dis[v] = dis[u] + w;
        }
    }
    for (int j = 1; j <= m; j++)
    {
        int u = e[j].u, v = e[j].v, w = e[j].w;
        if (dis[u] != INF && dis[v] > dis[u] + w)
            return 0;   // 负环
    }
    return 1;
}

int main()
{
    cin >> n >> m >> s;

#include <bits/stdc++.h>

using namespace std;

#define int long long

int n, m;
int a[100005];
int sum[400005], lazy[400005];

void pushup(int p)
{
    sum[p] = sum[p * 2] + sum[p * 2 + 1];
}

void pushdown(int p, int l, int r)
{
    if (lazy[p])
    {
        int mid = (l + r) / 2;
        sum[p * 2] += lazy[p] * (mid - l + 1);
        sum[p * 2 + 1] += lazy[p] * (r - mid);
        lazy[p * 2] += lazy[p];
        lazy[p * 2 + 1] += lazy[p];
        lazy[p] = 0;
    }
}

void build(int p, int l, int r)
{
    if (l == r)
    {
        sum[p] = a[l];
        return;
    }
    int mid = (l + r) / 2;
    build(p * 2, l, mid);
    build(p * 2 + 1, mid + 1, r);
    pushup(p);
}

void update(int p, int l, int r, int ql, int qr, int x)
{
    if (ql <= l && r <= qr)
    {
        sum[p] += x * (r - l + 1);
        lazy[p] += x;
        return;
    }
    pushdown(p, l, r);
    int mid = (l + r) / 2;
    if (ql <= mid)
        update(p * 2, l, mid, ql, qr, x);
    if (qr > mid)
        update(p * 2 + 1, mid + 1, r, ql, qr, x);
    pushup(p);
}

{
    if (ql <= l && r <= qr)
        return sum[p];
    pushdown(p, l, r);
    int mid = (l + r) / 2, res = 0;
    if (ql <= mid)
        res += query(p * 2, l, mid, ql, qr);

}

{
    sum[p] = (sum[p * 2] + sum[p * 2 + 1]) % mod;
}

void pushdown(int p, int l, int r)
{
    int mid = (l + r) / 2;
    sum[p * 2] = (sum[p * 2] * mul[p] + add[p] * (mid - l + 1)) % mod;
    sum[p * 2 + 1] = (sum[p * 2 + 1] * mul[p] + add[p] * (r - mid)) % mod;
    mul[p * 2] = mul[p * 2] * mul[p] % mod;
    mul[p * 2 + 1] = mul[p * 2 + 1] * mul[p] % mod;
    add[p * 2] = (add[p * 2] * mul[p] + add[p]) % mod;
    add[p * 2 + 1] = (add[p * 2 + 1] * mul[p] + add[p]) % mod;
    mul[p] = 1;
    add[p] = 0;
}

void build(int p, int l, int r)
{
    mul[p] = 1;
    if (l == r)
    {
        sum[p] = a[l] % mod;
        return;
    }
    int mid = (l + r) / 2;
    build(p * 2, l, mid);
    build(p * 2 + 1, mid + 1, r);
    pushup(p);
}

{
    if (ql <= l && r <= qr)
    {
        sum[p] = (sum[p] + x * (r - l + 1)) % mod;
        add[p] = (add[p] + x) % mod;
        return;
    }
    pushdown(p, l, r);
    int mid = (l + r) / 2;
    if (ql <= mid)
        update_add(p * 2, l, mid, ql, qr, x);
    if (qr > mid)
        update_add(p * 2 + 1, mid + 1, r, ql, qr, x);
    pushup(p);
}

void update_mul(int p, int l, int r, int ql, int qr, int x)
{
    if (ql <= l && r <= qr)
    {
        sum[p] = sum[p] * x % mod;
        add[p] = add[p] * x % mod;
        mul[p] = mul[p] * x % mod;
        return;
    }
    pushdown(p, l, r);
    int mid = (l + r) / 2;
    if (ql <= mid)
        update_mul(p * 2, l, mid, ql, qr, x);
    if (qr > mid)

    int mid = (l + r) / 2, res = 0;
    if (ql <= mid)
        res = (res + query(p * 2, l, mid, ql, qr)) % mod;
    if (qr > mid)
        res = (res + query(p * 2 + 1, mid + 1, r, ql, qr)) % mod;

}

void update(int &p, int pre, int l, int r, int x)
{
    p = ++tot;
    ls[p] = ls[pre];
    rs[p] = rs[pre];
    sum[p] = sum[pre] + 1;
    if (l == r)
        return;
    int mid = (l + r) / 2;
    if (x <= mid)
        update(ls[p], ls[pre], l, mid, x);
    else
        update(rs[p], rs[pre], mid + 1, r, x);
}

int query(int p, int q, int l, int r, int k)
{
    if (l == r)
        return l;
    int mid = (l + r) / 2;
    int cnt = sum[ls[q]] - sum[ls[p]];
    if (k <= cnt)
        return query(ls[p], ls[q], l, mid, k);
    return query(rs[p], rs[q], mid + 1, r, k - cnt);
}

    for (int i = 1; i <= n; i++)
    {

using namespace std;

int n, tot, rt;
int ch[100005][2], val[100005], rnd[100005], sz[100005];

int newnode(int x)
{
    val[++tot] = x;
    rnd[tot] = rand();
    sz[tot] = 1;
    return tot;
}

void pushup(int p)
{
    sz[p] = sz[ch[p][0]] + sz[ch[p][1]] + 1;
}

void split(int p, int x, int &a, int &b)
{
    if (!p)
    {
        a = b = 0;
        return;
    }
    if (val[p] <= x)
    {
        a = p;
        split(ch[p][1], x, ch[p][1], b);
    }
    else
    {
        b = p;
        split(ch[p][0], x, a, ch[p][0]);
    }
    pushup(p);
}

int merge(int a, int b)
{
    if (!a || !b)
        return a + b;
    if (rnd[a] < rnd[b])
    {
        ch[a][1] = merge(ch[a][1], b);
        pushup(a);
        return a;
    }
    else
    {
        ch[b][0] = merge(a, ch[b][0]);
        pushup(b);
        return b;
    }
}

void insert(int x)
{
    int a, b;
    split(rt, x, a, b);
    rt = merge(merge(a, newnode(x)), b);
}

void del(int x)
{
    int a, b, c;
    split(rt, x, a, b);
    split(a, x - 1, a, c);
    c = merge(ch[c][0], ch[c][1]);
    rt = merge(merge(a, c), b);
}

int kth(int p, int k)
{
    while (1)
    {
        if (k <= sz[ch[p][0]])
            p = ch[p][0];
        else if (k == sz[ch[p][0]] + 1)
            return val[p];
        else

            p = ch[p][1];
        }
    }
}

    return res;
}

#include <bits/stdc++.h>

using namespace std;

int n, tot, rt;
int ch[100005][2], fa[100005], val[100005], sz[100005], cnt[100005];

void pushup(int p)
{
    sz[p] = sz[ch[p][0]] + sz[ch[p][1]] + cnt[p];
}

int get(int p)
{
    return ch[fa[p]][1] == p;
}

void rotate(int x)
{
    int y = fa[x], z = fa[y], k = get(x);
    ch[y][k] = ch[x][k ^ 1];
    if (ch[x][k ^ 1])
        fa[ch[x][k ^ 1]] = y;
    ch[x][k ^ 1] = y;
    fa[y] = x;
    fa[x] = z;
    if (z)
        ch[z][get(y)] = x;
    pushup(y);
    pushup(x);
}

void splay(int x, int goal)
{
    while (fa[x] != goal)
    {
        int y = fa[x], z = fa[y];
        if (z != goal)
        {
            if (get(x) == get(y))
                rotate(y);
            else
                rotate(x);
        }
        rotate(x);
    }
    if (!goal)
        rt = x;
}

void insert(int x)
{
    if (!rt)
    {
        rt = ++tot;
        val[rt] = x;
        cnt[rt] = sz[rt] = 1;
        return;
    }
    int p = rt, f = 0;
    while (1)
    {
        if (val[p] == x)
        {
            cnt[p]++;
            sz[p]++;
            splay(p, 0);
            return;
        }
        f = p;
        p = ch[p][val[p] < x];
        if (!p)
        {
            p = ++tot;
            val[p] = x;
            cnt[p] = sz[p] = 1;
            fa[p] = f;
            ch[f][val[f] < x] = p;
            splay(p, 0);
            return;
        }
    }
}

{
    int p = rt;
    while (1)
    {
        if (ch[p][0] && k <= sz[ch[p][0]])
            p = ch[p][0];
        else if (k > sz[ch[p][0]] + cnt[p])
        {
            k -= sz[ch[p][0]] + cnt[p];
            p = ch[p][1];
        }
        else
            return val[p];
    }
}

{

    while (1)
    {
        if (val[p] > x)
            p = ch[p][0];
        else
        {
            res += sz[ch[p][0]];
            if (val[p] == x)
            {
                splay(p, 0);
                return res + 1;
            }
            res += cnt[p];
            p = ch[p][1];
        }
    }
}

}

    while (ch[p][0])
        p = ch[p][0];
    int res = val[p];
    return res;
}

#include <bits/stdc++.h>

using namespace std;

int n, m;
int a[100005];
int f[100005][20];

void build()
{
    for (int i = 1; i <= n; i++)
    {
        f[i][0] = a[i];
    }
    int k = log2(n);
    for (int j = 1; j <= k; j++)
    {
        for (int i = 1; i + (1 << j) - 1 <= n; i++)
        {
            f[i][j] = max(f[i][j - 1], f[i + (1 << (j - 1))][j - 1]);
        }
    }
}

int query(int l, int r)
{
    int k = log2(r - l + 1);
    return max(f[l][k], f[r - (1 << k) + 1][k]);
}

int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    build();
    while (m--)
    {
        int l, r;
        cin >> l >> r;
        cout << query(l, r) << '\n';
    }
    return 0;
}
